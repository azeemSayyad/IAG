# Step 0.5 — Core Database Tables

## Table 1: tenants

Top-level organization. Each company subscribing to the platform is a tenant.

```sql
CREATE TABLE tenants (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name VARCHAR(255) NOT NULL,
    subscription_plan VARCHAR(50) NOT NULL DEFAULT 'starter',
        -- enum: starter, growth, enterprise
    status VARCHAR(50) NOT NULL DEFAULT 'active',
        -- enum: active, suspended, cancelled, trial
    max_agents INT NOT NULL DEFAULT 5,
    max_leads_per_month INT NOT NULL DEFAULT 1000,
    settings JSONB DEFAULT '{}',
        -- stores: business_hours, timezone, booking_rules, ai_tone
    created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    deleted_at TIMESTAMPTZ DEFAULT NULL
);

CREATE INDEX idx_tenants_status ON tenants(status) WHERE deleted_at IS NULL;
```

---

## Table 2: users

Authentication and identity for all human users.

```sql
CREATE TABLE users (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    tenant_id UUID NOT NULL REFERENCES tenants(id),
    email VARCHAR(255) NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    first_name VARCHAR(100) NOT NULL,
    last_name VARCHAR(100) NOT NULL,
    role VARCHAR(50) NOT NULL DEFAULT 'agent',
        -- enum: super_admin, tenant_admin, manager, agent
    status VARCHAR(50) NOT NULL DEFAULT 'active',
        -- enum: active, inactive, locked, invited
    last_login_at TIMESTAMPTZ,
    created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    deleted_at TIMESTAMPTZ DEFAULT NULL,

    CONSTRAINT uq_users_email UNIQUE (email)
);

CREATE INDEX idx_users_tenant ON users(tenant_id) WHERE deleted_at IS NULL;
CREATE INDEX idx_users_email ON users(email) WHERE deleted_at IS NULL;
CREATE INDEX idx_users_role ON users(tenant_id, role) WHERE deleted_at IS NULL;
```

---

## Table 3: agents

Agent-specific data. One-to-one with users where role = 'agent'.

```sql
CREATE TABLE agents (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    tenant_id UUID NOT NULL REFERENCES tenants(id),
    user_id UUID NOT NULL REFERENCES users(id),
    timezone VARCHAR(50) NOT NULL DEFAULT 'America/New_York',
    daily_capacity INT NOT NULL DEFAULT 8,
        -- max appointments per day
    max_concurrent INT NOT NULL DEFAULT 1,
        -- max simultaneous appointments
    skills JSONB DEFAULT '[]',
        -- ["auto", "home", "life", "commercial"]
    weight INT NOT NULL DEFAULT 100,
        -- for weighted round-robin assignment (higher = more appointments)
    status VARCHAR(50) NOT NULL DEFAULT 'active',
        -- enum: active, inactive, on_break, offline
    created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT now(),

    CONSTRAINT uq_agents_user UNIQUE (user_id)
);

CREATE INDEX idx_agents_tenant ON agents(tenant_id);
CREATE INDEX idx_agents_status ON agents(tenant_id, status);
```

## Table 4: leads

Every potential customer in the system.

```sql
CREATE TABLE leads (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    tenant_id UUID NOT NULL REFERENCES tenants(id),
    source VARCHAR(100) NOT NULL,
        -- enum: csv_import, webhook, api, manual, facebook, google, referral
    first_name VARCHAR(100) NOT NULL,
    last_name VARCHAR(100) NOT NULL,
    phone VARCHAR(20) NOT NULL,
    email VARCHAR(255),
    state VARCHAR(50),
    city VARCHAR(100),
    zip_code VARCHAR(10),
    lead_score INT DEFAULT 0,
        -- 0-100, computed by lead scoring ML
    status VARCHAR(50) NOT NULL DEFAULT 'new',
        -- enum: new, contacted, replied, qualified, booked,
        --       completed, unqualified, no_show, nurture
    campaign_id UUID,
    tags JSONB DEFAULT '[]',
    custom_fields JSONB DEFAULT '{}',
    last_contacted_at TIMESTAMPTZ,
    created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    deleted_at TIMESTAMPTZ DEFAULT NULL
);

CREATE INDEX idx_leads_tenant ON leads(tenant_id) WHERE deleted_at IS NULL;
CREATE INDEX idx_leads_status ON leads(tenant_id, status) WHERE deleted_at IS NULL;
CREATE INDEX idx_leads_phone ON leads(tenant_id, phone) WHERE deleted_at IS NULL;
CREATE INDEX idx_leads_email ON leads(tenant_id, email) WHERE deleted_at IS NULL;
CREATE INDEX idx_leads_score ON leads(tenant_id, lead_score DESC) WHERE deleted_at IS NULL;
CREATE INDEX idx_leads_campaign ON leads(tenant_id, campaign_id) WHERE deleted_at IS NULL;
```

## Table 5: conversations

Tracks the AI conversation thread with each lead.

```sql
CREATE TABLE conversations (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    tenant_id UUID NOT NULL REFERENCES tenants(id),
    lead_id UUID NOT NULL REFERENCES leads(id),
    status VARCHAR(50) NOT NULL DEFAULT 'initiated',
        -- enum: initiated, active, booking, booked, paused, stopped, closed
    intent VARCHAR(50),
        -- latest detected intent: POSITIVE, INTERESTED, SKEPTICAL,
        -- NEGATIVE, STOP, BOOK_NOW, QUESTION, RESCHEDULE
    sentiment VARCHAR(50),
        -- latest sentiment: positive, neutral, negative
    message_count INT NOT NULL DEFAULT 0,
    last_message_at TIMESTAMPTZ,
    last_message_from VARCHAR(20),
        -- "customer" or "ai"
    ai_context JSONB DEFAULT '{}',
        -- stores: memory, objections, preferences, tone state
    created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE INDEX idx_conversations_tenant ON conversations(tenant_id);
CREATE INDEX idx_conversations_lead ON conversations(lead_id);
CREATE INDEX idx_conversations_status ON conversations(tenant_id, status);
CREATE INDEX idx_conversations_intent ON conversations(tenant_id, intent);
CREATE INDEX idx_conversations_last_msg ON conversations(tenant_id, last_message_at DESC);
```

## Table 6: messages

Individual messages within a conversation.

```sql
CREATE TABLE messages (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    conversation_id UUID NOT NULL REFERENCES conversations(id),
    tenant_id UUID NOT NULL REFERENCES tenants(id),
    sender VARCHAR(20) NOT NULL,
        -- "customer" or "ai"
    content TEXT NOT NULL,
    message_type VARCHAR(50) NOT NULL DEFAULT 'sms',
        -- enum: sms, system, notification
    intent VARCHAR(50),
        -- detected intent for this specific message
    sentiment VARCHAR(50),
        -- detected sentiment for this specific message
    metadata JSONB DEFAULT '{}',
        -- stores: provider_message_sid, delivery_status, error, prompt_used
    created_at TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE INDEX idx_messages_conversation ON messages(conversation_id, created_at);
CREATE INDEX idx_messages_tenant ON messages(tenant_id, created_at);
```

## Table 7: appointments

Booked call slots between a lead and an agent.

```sql
CREATE TABLE appointments (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    tenant_id UUID NOT NULL REFERENCES tenants(id),
    lead_id UUID NOT NULL REFERENCES leads(id),
    agent_id UUID NOT NULL REFERENCES agents(id),
    conversation_id UUID REFERENCES conversations(id),
    start_time TIMESTAMPTZ NOT NULL,
    end_time TIMESTAMPTZ NOT NULL,
    status VARCHAR(50) NOT NULL DEFAULT 'confirmed',
        -- enum: pending, confirmed, completed, cancelled, no_show, rescheduled
    disposition VARCHAR(50),
        -- set after call: won, lost, follow_up, no_answer
    notes TEXT,
    call_duration_seconds INT,
    reminder_24h_sent BOOLEAN DEFAULT FALSE,
    reminder_1h_sent BOOLEAN DEFAULT FALSE,
    reminder_15m_sent BOOLEAN DEFAULT FALSE,
    cancelled_reason VARCHAR(255),
    rescheduled_from UUID REFERENCES appointments(id),
    created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE INDEX idx_appointments_tenant ON appointments(tenant_id);
CREATE INDEX idx_appointments_agent ON appointments(agent_id, start_time);
CREATE INDEX idx_appointments_lead ON appointments(lead_id);
CREATE INDEX idx_appointments_status ON appointments(tenant_id, status);
CREATE INDEX idx_appointments_time ON appointments(tenant_id, start_time);
CREATE INDEX idx_appointments_reminders ON appointments(status, start_time)
    WHERE status = 'confirmed'
    AND (reminder_24h_sent = FALSE OR reminder_1h_sent = FALSE OR reminder_15m_sent = FALSE);
```

## Table 8: agent_availability

Tracks when agents are available for appointments.

```sql
CREATE TABLE agent_availability (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    agent_id UUID NOT NULL REFERENCES agents(id),
    tenant_id UUID NOT NULL REFERENCES tenants(id),
    start_time TIMESTAMPTZ NOT NULL,
    end_time TIMESTAMPTZ NOT NULL,
    availability_status VARCHAR(50) NOT NULL DEFAULT 'available',
        -- enum: available, booked, break, offline, holiday
    recurrence_rule VARCHAR(255),
        -- iCal RRULE format for recurring schedules
        -- e.g., "FREQ=WEEKLY;BYDAY=MO,TU,WE,TH,FR"
    notes VARCHAR(255),
    created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE INDEX idx_availability_agent ON agent_availability(agent_id, start_time);
CREATE INDEX idx_availability_tenant ON agent_availability(tenant_id, start_time);
CREATE INDEX idx_availability_status ON agent_availability(tenant_id, availability_status, start_time);
```
